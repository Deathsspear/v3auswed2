	}

	}

	scripted_button = je_chaebol_control_gromwell_purple

	scripted_button = je_chaebol_control_ginseng

	scripted_button = je_chaebol_control_koksan_leaf

	scripted_button = je_chaebol_control_dragonfly_robes

	scripted_button = je_chaebol_control_siwidae_cannons


	complete = {
		custom_tooltip = {
			text = anthracite_controlled
			has_variable = je_chaebol_control_anthracite	
		}
		custom_tooltip = {
			text = living_engines_controlled
			has_variable = je_chaebol_control_living_engines	
		}

			text = ginseng_controlled

		}
		custom_tooltip = {
			text = dragonfly_robes_controlled
			has_variable = je_chaebol_control_dragonfly_robes	
		}
		custom_tooltip = {
			text = siwidae_cannons_controlled
			has_variable = je_chaebol_control_siwidae_cannons	
		}

		set_variable = chaebol_risen

			remove_modifier = the_chaebols_anthracite_interest
		}
		if = {
			limit = { has_modifier = the_chaebols_living_engines_interest }
			remove_modifier = the_chaebols_living_engines_interest
		}
		if = {
			limit = { has_modifier = the_chaebols_gromwell_purple_interest }
			remove_modifier = the_chaebols_gromwell_purple_interest
		}
		if = {
			limit = { has_modifier = the_chaebols_ginseng_interest }
			remove_modifier = the_chaebols_ginseng_interest
		}
		if = {
			limit = { has_modifier = the_chaebols_koksan_leaf_interest }
			remove_modifier = the_chaebols_koksan_leaf_interest
		}
		if = {
			limit = { has_modifier = the_chaebols_leviathans_balm_interest }
			remove_modifier = the_chaebols_leviathans_balm_interest
		}
		if = {
			limit = { has_modifier = the_chaebols_dragonfly_robes_interest }
			remove_modifier = the_chaebols_dragonfly_robes_interest
		}
		if = {
			limit = { has_modifier = the_chaebols_siwidae_cannons_interest }
			remove_modifier = the_chaebols_siwidae_cannons_interest
		}
		if = {
			limit = { has_modifier = the_chaebols_established_modifier }
			remove_modifier = the_chaebols_established_modifier
		}
		if = {
			limit = { has_modifier = the_chaebols_inefficient_modifier }
			remove_modifier = the_chaebols_inefficient_modifier
		}
		add_modifier = {
			name = the_chaebols_subdued
		}
	}

	timeout = 7300
	on_timeout = { id = chaebol_event.4 popup = yes }

	}
