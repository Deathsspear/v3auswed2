        }
    }

            c:CIR ?= {

                }
            }
            c:CHC ?= {
                is_subject = yes
                NOT = {
                    is_subject_of = ROOT
                }
            }
        }
    }

        set_global_variable = caucasus_failure_var
    }

    on_yearly_pulse = {
        random_events = {
            90 = 0
            10 = caucasuswar.1
            10 = caucasuswar.2
            10 = caucasuswar.3
            10 = caucasuswar.4
            50 = caucasuswar.6
        }
    }

    weight = 100

    	
	should_be_pinned_by_default_uninvolved_or_context = yes
