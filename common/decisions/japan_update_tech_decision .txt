japan_update_tech_decision = {
	is_shown = {
		is_player = yes
		OR = {
			country_has_primary_culture = cu:japanese
		}
		OR = {
			NOT = { has_technology_researched = atmospheric_engine }
			NOT = { has_technology_researched = stock_exchange }
			NOT = { has_technology_researched = corporate_charters }
			NOT = { has_technology_researched = mandatory_service }
			NOT = { has_technology_researched = line_infantry }
			NOT = { has_technology_researched = romanticism }
			NOT = { has_technology_researched = empiricism }
		}
	}

	possible = {
		is_player = yes
		OR = {
			NOT = { has_technology_researched = atmospheric_engine }
			NOT = { has_technology_researched = stock_exchange }
			NOT = { has_technology_researched = corporate_charters }
			NOT = { has_technology_researched = mandatory_service }
			NOT = { has_technology_researched = line_infantry }
			NOT = { has_technology_researched = romanticism }
			NOT = { has_technology_researched = empiricism }
		}
	}

	when_taken = {
		#Production
		add_technology_researched = atmospheric_engine

		#Military
		add_technology_researched = mandatory_service
		add_technology_researched = line_infantry

		#Society
		add_technology_researched = stock_exchange
		add_technology_researched = corporate_charters
		add_technology_researched = romanticism
		add_technology_researched = empiricism
	}
}
