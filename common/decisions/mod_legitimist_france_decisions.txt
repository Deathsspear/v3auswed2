mod_enact_slavery= {
	is_shown = {
		#always = no
		is_player = yes
		c:FRA ?= THIS
		NOT = {
			OR = {
				has_law = law_type:law_slave_trade
				has_variable = enacted_slavery
			}
		}
	}

	possible = {
		ruler = {has_ideology = ideology:ideology_legitimist}
		#always = no
		#is_player = yes
	}
	when_taken = {
		set_global_variable = enacted_slavery
		activate_law = law_type:law_slave_trade
	}
}
