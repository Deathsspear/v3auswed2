building_investment_building_allowed = {
	# Government/private construction without an investor_country is domestic.
	trigger_if = {
		limit = {
			NOT = { exists = scope:investor_country }
		}
		always = yes
	}
	trigger_else = {
		owner = {
			save_temporary_scope_as = building_investment_state_owner
		}
		trigger_if = {
			limit = {
				exists = scope:building_investment_state_owner.top_overlord
			}
			scope:building_investment_state_owner.top_overlord = {
				save_temporary_scope_as = building_investment_top_overlord
			}
		}
		OR = {
			# Domestic construction.
			owner = scope:investor_country

			# Vanilla treaty foreign investment rights.
			scope:investor_country = {
				has_treaty_foreign_investment_rights_with = {
					TARGET = scope:building_investment_state_owner
				}
			}

			# An overlord may invest in its subject.
			scope:building_investment_state_owner = {
				is_subject_of = scope:investor_country
			}

			# Rights granted by a subject's top overlord extend to the subject.
			AND = {
				scope:building_investment_top_overlord ?= {
					scope:investor_country = {
						has_treaty_foreign_investment_rights_with = {
							TARGET = scope:building_investment_top_overlord
						}
					}
				}
			}

			# Subject-to-overlord investment requires the vanilla diplomatic pact.
			AND = {
				scope:investor_country = {
					is_subject_of = scope:building_investment_state_owner
					any_scope_diplomatic_pact = {
						OR = {
							is_diplomatic_action_type = da_overlord_grant_investment_rights
							is_diplomatic_action_type = da_subject_request_investment_rights
						}
						"diplomatic_pact_other_country(scope:investor_country)" = scope:building_investment_state_owner
					}
				}
			}

			# Vanilla power-bloc foreign investment principle.
			AND = {
				scope:investor_country.power_bloc ?= {
					has_principle = principle:principle_foreign_investment_3
					any_power_bloc_member = {
						this = scope:building_investment_state_owner
					}
					scope:building_investment_state_owner = {
						country_rank > scope:investor_country.country_rank
					}
				}
			}

			# This mod: the active article must name this exact building type.
			scope:building_investment_state_owner = {
				any_scope_treaty = {
					binds = scope:investor_country
					any_scope_article = {
						has_type = building_investment_rights
						source_country = scope:building_investment_state_owner
						target_country = scope:investor_country
						input_building_type = $BUILDING_TYPE$
					}
				}
			}
		}
	}
}
