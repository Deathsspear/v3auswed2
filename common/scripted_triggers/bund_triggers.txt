# German Confederation (Bund) triggers
# Items marked VERIFY use names I could not confirm against vanilla files. The game
# reports unknown keys in error.log on startup, so each one is cheap to check.

# Country scope. Membership comes from vanilla (country_is_in_german_confederation is used by
# vanilla je_grunderzeit), minus anyone expelled by this mod.
bund_is_member = {
	country_is_in_german_confederation = yes
	NOT = { has_variable = bund_expelled }
	NOT = { has_global_variable = bund_collapsed }
}

# State scope. True for land that counts as Federal Territory: a state inside vanilla's
# geographic_region_german_confederation that is owned by a current member.
# That region covers the Bund parts of Austria (Austria, Styria, Tyrol, South Tyrol, Bohemia,
# Moravia, Istria, Slovenia) and Prussia, and leaves out Hungary, Galicia, Lombardy-Venetia,
# Dalmatia, East/West Prussia and Posen, so a play for those never unlocks the button.
# is_in_geographic_region is used at state scope by vanilla JE map_color blocks.
bund_state_is_federal_territory = {
	owner ?= { bund_is_member = yes }
	is_in_geographic_region = geographic_region_german_confederation
}

# State scope. Owned by a member but outside the Confederation (Austria's Hungary, Galicia,
# Lombardy-Venetia, Dalmatia etc.; Prussia's East/West Prussia and Posen). Drawn as cross-hatching.
bund_state_is_member_non_federal = {
	owner ?= { bund_is_member = yes }
	NOT = { bund_state_is_federal_territory = yes }
}

# Diplomatic play scope. Is this a play the member (scope:bund_caller) may answer?
# Needs scope:bund_caller saved by the caller.
# is_war, target_is and the initiator scope are all documented for the play scope.
bund_play_is_answerable = {
	is_war = no
	target_is = scope:bund_caller
	initiator = {
		save_temporary_scope_as = bund_aggressor
		NOT = { bund_is_member = yes }
	}
	# The aggressor must hold a war goal against Federal Territory owned by the defender.
	# Wiki: is_target_of_wargoal is a state trigger, "target of any war goal in wars involving"
	# the given country. Its wiki syntax example reads has_war_goal = <country>, so confirm the
	# exact key with script_docs. Also confirm it counts a play that has not become a war yet.
	scope:bund_caller = {
		any_scope_state = {
			bund_state_is_federal_territory = yes
			is_target_of_wargoal = scope:bund_aggressor
		}
	}
}

# Country scope (a member). Does this member have an answerable play right now?
bund_has_answerable_play = {
	save_temporary_scope_as = bund_caller
	is_diplomatic_play_target = yes
	# any_diplomatic_play is a documented global iterator (scope none)
	any_diplomatic_play = {
		bund_play_is_answerable = yes
	}
}
