# An Offer for Louisiana - USA EVENT
louisiana.1 = {
	type = country_event
	placement = ROOT

	title = louisiana.1.t
	desc = louisiana.1.d
	flavor = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:louisiana_seller = {
						c:USA ?= this
					}
				}
				desc = louisiana.1.f_ru
			}
			triggered_desc = {
				trigger = {
					always = yes
				}
				desc = louisiana.1.f
			}
		}
	}
	
	cooldown = { days = 180 }

	duration = 3

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	trigger = {
	}

	immediate = {
		# Scopes set in louisiana.4
	}

	option = {
		name = louisiana.1.a
		add_modifier = {
			name = alaska_rus_big_money
			months = 36
			is_decaying = no
		}
		set_variable = gave_up_on_west #disables manifest destiny decision
		scope:louisiana_buyer = {
			trigger_event = { id = louisiana.2 days = 0 popup = yes }
		}
		c:FRA = {create_bidirectional_truce = {country = c:USA months = 120}}
		ai_chance = {
			base = 75
		}
	}
	option = {
		name = louisiana.1.b
		default_option = yes
		scope:louisiana_buyer = {
			trigger_event = { id = louisiana.3 days = 0 popup = yes }
		}
		show_as_tooltip = {
			change_relations = {
				country = scope:louisiana_buyer
				value = -20
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					scope:louisiana_seller = {
						NOT = {
							has_variable = louisiana_sale_cooldown #Unsolicited offer
						}
					}
				}
				add = 25
			}
		}
	}
}

# Making an Offer - FRA EVENT
louisiana.4 = {
	type = country_event
	placement = ROOT

	title = louisiana.4.t
	desc = louisiana.4.d
	flavor = louisiana.4.f

	duration = 3

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	trigger = {
	}

	immediate = {
		root = {
			save_scope_as = louisiana_buyer
		}
		s:STATE_LOUISIANA = {
			random_scope_state = {
				limit = {
					owner = {
						NOT = {
							is_country_type = decentralized
						}
					}
				}
				save_scope_as = louisiana_scope
				if = { # For selling Russian-American Company
					limit = {
						owner = {
							is_subject = yes
							capital = {
								state_region = s:STATE_LOUISIANA
							}
						}
					}
					owner.top_overlord = {
						save_scope_as = louisiana_seller
					}
				}
				else = { # For other cases
					owner = {
						save_scope_as = louisiana_seller
					}
				}
			}
		}
	}
	
	cooldown = { days = 180 }

	option = {
		name = louisiana.4.a
		ai_chance = {
			base = 1
		}
		custom_tooltip = louisiana.4.tt1 # 3 million over a year for five years
		scope:louisiana_seller ?= {
			trigger_event = {
				id = louisiana.1
				popup = yes
			}
		}
	}
	option = { # perhaps another time
		name = louisiana.4.c
		default_option = yes
		ai_chance = {
			base = 0
		}
		scope:louisiana_seller = {
			post_notification = louisiana_negotiations_fail
		}
	}
}
