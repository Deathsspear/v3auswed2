namespace = chaebol_event

chaebol_event.1 = {
	type = country_event
	placement = ROOT

	trigger = {
		exists = c:KOR
		this = c:KOR
	}

	title = chaebol_event.1.t
	desc = chaebol_event.1.d
	flavor = chaebol_event.1.f

	event_image = {
		video = "gfx/event_pictures/unspecific_signed_contract.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/unspecific/signed_contract"

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3
	
	option = {
		name = chaebol_event.1.a #the rise of the chaebols
		default_option = yes
		add_modifier = {
			name = the_chaebols_modifier
		}
		add_modifier = {
			name = the_chaebols_inefficient_modifier
		}
		add_modifier = {
			name = the_chaebols_landowners_happy_modifier
		}			
		ig:ig_landowners = {
            remove_ideology = ideology_paternalistic 
            add_ideology = ideology_chaebol_paternalistic
        }
		ig:ig_landowners = {
            remove_ideology = ideology_hierarchic 
            add_ideology = ideology_chaebol_hierarchic
        }
	}
}
